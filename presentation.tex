\begin{frame}{1 \textperiodcentered{} Dos aciertos complementarios, una externalidad}
\small
\textbf{Pregunta.} ¿Puede mejor consejo privado destruir el conocimiento público que lo vuelve útil?
\[
\max_{e\ge0}\ U=f(0,0)+G(X)\Delta_G+G(X)G(Y)\Delta_X-\frac{e^\alpha}{\alpha},
\quad Y=\sigma^{-2}+\lambda_Ie+\tau_A.
\]
\[
\underbrace{\Delta_XG(X)\lambda_Ig(Y)}_{MB(e;X,\tau_A)}=e^{\alpha-1}.
\]
\begin{itemize}\item $X$: precisión general heredada; $\tau_A$: precisión agéntica contextual.
\item El esfuerzo produce señal privada y una señal pública que el agente atomístico no internaliza.
\item \,\textbf{Assumption 1:} $\Delta_I=0$, $\Delta_X>0$ (el contexto solo no produce valor).
\end{itemize}
\sourcefoot{PDF 20-02-2026, §§3.1--3.5, pp. 8--15; ecuaciones (4)--(6).}
\end{frame}

\begin{frame}{2 \textperiodcentered{} Observación 1: interior y condicional}
\small
Con gaussianidad, $\lambda_I>0$, $\alpha>1$, $\Delta_I=0$, $\Delta_X>0$, $X,Y>0$:
\[
U_{eX}=\Delta_X\lambda_Ig(X)g(Y)>0,
\qquad U_{e\tau_A}=\Delta_XG(X)\lambda_Ig'(Y)<0.
\]
\begin{block}{Qué hace cada supuesto}
$g>0$ y $\Delta_X>0$ dan complementariedad; $g'<0$ da sustitución; costo convexo da esfuerzo único. Topkis implica $e_X>0$ y $e_{\tau_A}<0$ para $X>0$.
\end{block}
\keybox{\textbf{Cota olvidada en el enunciado estricto:} en $X=0$, $e^*=0$ para todo $\tau_A$ y $g(X)$ diverge; la derivada cruzada no es una desigualdad finita.}
\sourcefoot{§§3.4--3.5, pp. 13--15; Observaciones 1--2 y nota 4.}
\end{frame}

\begin{frame}{4 \textperiodcentered{} Donde no le creí: cero no es robusto}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{.61\textwidth}
\begin{enumerate}\item El candidato inicial era falso: §§5.1--5.2 sí permiten agregación por IA y datos sintéticos.
\item Si $\tau_{syn}>0$, entonces $F_{syn}(0)>0$: cero deja de ser fijo (Prop. 15).
\item Si $\Delta_I>0$ o queda una masa de especialistas, hay retorno/esfuerzo positivo aun con $X=0$.
\item Quispe--Xu: expansión persistente de lenguajes sin commits del agente es compatible con aprendizaje, canal ausente aquí.
\end{enumerate}
\keybox{\textbf{Veredicto.} Es defendible un estado de conocimiento bajo. El colapso exactamente a cero exige $\Delta_I=0$ y flujo autónomo nulo.}
